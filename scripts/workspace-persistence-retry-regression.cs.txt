// Standalone regression of production persistence and lifetime coordination.
using FaceShield.Services.Workspace;
using FaceShield.Services.Video;
using FaceShield.ViewModels.Workspace;
using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Threading.Tasks;

var store = new WorkspaceStateStore();
var masks = new FrameMaskProvider();
using (var coordinator = new WorkspacePersistenceCoordinator(store, masks))
{
    store.FailNextSave = true;
    Expect<IOException>(() => coordinator.SaveNow(() => new WorkspaceSnapshot
    {
        VideoPath = "video.mp4", Sequence = 1
    }), "Expected terminal save failure");

    // A transient I/O failure must not poison later SaveNow or QueueSaveAsync.
    coordinator.SaveNow(() => new WorkspaceSnapshot
    {
        VideoPath = "video.mp4", Sequence = 2
    });
    await coordinator.QueueSaveAsync(() => new WorkspaceSnapshot
    {
        VideoPath = "video.mp4", Sequence = 3
    });
    await coordinator.FlushAsync();
    Check(store.Saved.SequenceEqual(new[] { 2, 3 }),
        "Failed terminal save blocked retry or subsequent queued save");

    Expect<InvalidOperationException>(() =>
        coordinator.SaveNow(() => throw new InvalidOperationException("capture failed")),
        "Expected snapshot-capture failure");
    coordinator.SaveNow(() => new WorkspaceSnapshot
    {
        VideoPath = "video.mp4", Sequence = 4
    });
    Check(store.Saved.Last() == 4,
        "Snapshot capture failure poisoned later final save");
}

var lifetime = new WorkspaceOperationLifetime(() => { });
Check(lifetime.TryBeginExclusiveProcessing(), "Could not begin Auto operation");
Task idle = lifetime.WaitForIdleAsync();
Check(!idle.IsCompleted, "Idle waiter completed while Auto was active");
Check(!lifetime.TryCloseAdmissionWhenIdle(), "Eviction admitted an active operation");
Check(lifetime.CloseAdmission(), "Could not close admission");
Check(!lifetime.TryBegin(), "Closing admission allowed another operation");
lifetime.EndExclusiveProcessing();
await idle;
lifetime.ReopenAdmission();
Check(lifetime.TryBegin(), "Failed close prevented new work after reopening");
lifetime.End();
Check(lifetime.TryCloseAdmissionWhenIdle(), "Idle cache eviction could not close admission");
Check(!lifetime.TryBegin(), "Evicted workspace still accepted work");
lifetime.ReopenAdmission();
Check(lifetime.TryBegin(), "Failed eviction could not reopen admission");
lifetime.End();
// Exercise the same production registration helper used by BOTH manual track
// commands. Hold the admission callback after the lifetime is acquired while
// a concurrent shutdown closes admission and invokes its cancellation event.
var trackingLifetime = new WorkspaceOperationLifetime(() => { });
object trackingCommitGate = new();
CancellationTokenSource? activeTracking = null;
using var trackingSource = new CancellationTokenSource();
using var registered = new System.Threading.ManualResetEventSlim(false);
using var releaseRegistration = new System.Threading.ManualResetEventSlim(false);
using var closeRequested = new System.Threading.ManualResetEventSlim(false);
trackingLifetime.AdmissionClosed += () =>
{
    lock (trackingCommitGate)
    {
        activeTracking?.Cancel();
    }
};

Task<bool> registration = Task.Run(() => ManualTrackingCancellationGate.TryBegin(
    trackingCommitGate, ref activeTracking, trackingSource, () =>
    {
        bool acquired = trackingLifetime.TryBeginExclusiveProcessing();
        Check(ReferenceEquals(activeTracking, trackingSource),
            "Lifetime admission started before cancellation source was visible");
        registered.Set();
        if (!releaseRegistration.Wait(TimeSpan.FromSeconds(10)))
            throw new TimeoutException("Admission registration callback stalled");
        return acquired;
    }));
Check(registered.Wait(TimeSpan.FromSeconds(10)),
    "Tracking registration did not reach the admission checkpoint");
Task<bool> closing = Task.Run(() =>
{
    closeRequested.Set();
    return trackingLifetime.CloseAdmission();
});
Check(closeRequested.Wait(TimeSpan.FromSeconds(10)), "Close callback did not start");
releaseRegistration.Set();
Check(await registration.WaitAsync(TimeSpan.FromSeconds(10)),
    "Lifetime rejected a tracking operation already admitted");
Check(await closing.WaitAsync(TimeSpan.FromSeconds(10)),
    "Concurrent shutdown failed to close admission");
Check(trackingSource.IsCancellationRequested,
    "Shutdown missed cancellation during tracking source registration");
Task trackingIdle = trackingLifetime.WaitForIdleAsync();
Check(!trackingIdle.IsCompleted,
    "Close did not wait for the canceled tracking lifetime to drain");
ManualTrackingCancellationGate.Finish(
    trackingCommitGate, ref activeTracking, trackingSource);
trackingLifetime.EndExclusiveProcessing();
await trackingIdle.WaitAsync(TimeSpan.FromSeconds(10));
Check(activeTracking == null, "Tracking source remained registered after completion");

// Opposite ordering: shutdown wins before the token can be registered.
using var rejectedSource = new CancellationTokenSource();
Check(!ManualTrackingCancellationGate.TryBegin(
        trackingCommitGate, ref activeTracking, rejectedSource,
        trackingLifetime.TryBeginExclusiveProcessing),
    "Tracking was admitted after shutdown closed admission");
Check(activeTracking == null, "Rejected tracking left a stale source");
ManualTrackingCancellationGate.Finish(
    trackingCommitGate, ref activeTracking, rejectedSource);

// Failure in lifetime admission must clear the published source, too.
using var failingSource = new CancellationTokenSource();
Expect<InvalidOperationException>(() => ManualTrackingCancellationGate.TryBegin(
    trackingCommitGate, ref activeTracking, failingSource,
    () => throw new InvalidOperationException("simulated admission failure")),
    "Lifetime callback failure was not propagated");
Check(activeTracking == null, "Admission exception left a leaked tracking source");
ManualTrackingCancellationGate.Finish(
    trackingCommitGate, ref activeTracking, failingSource);

Console.WriteLine("PASS: manual tracking token-before-lifetime admission, concurrent shutdown cancel, denied and failed admission");
Console.WriteLine("PASS: retryable SaveNow/QueueSave, failed capture, closing drain, idle-only eviction");

static void Check(bool value, string message)
{
    if (!value) throw new InvalidOperationException(message);
}
static void Expect<T>(Action action, string message) where T : Exception
{
    try { action(); }
    catch (T) { return; }
    throw new InvalidOperationException(message);
}

namespace FaceShield.Services.Video
{
    public sealed class FrameMaskProvider
    {
        public sealed class PersistenceSnapshot : IDisposable
        {
            public void Dispose() { }
        }

        public PersistenceSnapshot CreatePersistenceSnapshot() => new();
    }
}

namespace FaceShield.Services.Workspace
{
    public sealed class WorkspaceSnapshot
    {
        public string VideoPath { get; init; } = "";
        public int Sequence { get; init; }
    }

    public sealed class WorkspaceStateStore
    {
        public sealed class SaveLease : IDisposable
        {
            public void Dispose() { }
        }

        public bool FailNextSave { get; set; }
        public List<int> Saved { get; } = new();
        public SaveLease AcquireWorkspaceSaveLease(string path) => new();
        public void SaveWorkspaceSnapshot(WorkspaceSnapshot snapshot,
            FaceShield.Services.Video.FrameMaskProvider.PersistenceSnapshot masks,
            SaveLease lease)
        {
            if (FailNextSave)
            {
                FailNextSave = false;
                throw new IOException("simulated transient disk failure");
            }
            Saved.Add(snapshot.Sequence);
        }
    }
}

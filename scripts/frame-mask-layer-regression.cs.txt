// unchanged manual source, including in the export snapshot and on disk.
string retryVideo = Path.GetTempFileName();
try
{
    using var retryProvider = new FrameMaskProvider();
    var source = new WriteableBitmap(size, new Vector(96, 96),
        PixelFormat.Bgra8888, AlphaFormat.Premul);
    PaintRect(source, 60, 30, 12, 12, 255);
    retryProvider.SetIndependentManualMask(42, source);
    ManualMaskKeyframeTimeline.Configure(retryProvider, enabled: true, videoPath: retryVideo);
    Assert(retryProvider.TryCloneStoredMask(42, out var sourceCopy),
        "Missing manual source for retry regression");
    string fingerprint;
    using (sourceCopy)
        fingerprint = ManualMaskFingerprint.Compute(sourceCopy);

    ManualMaskTrackSegment MakeSegment(int frames, double offsetX = 0, bool failed = false)
    {
        var component = new ManualMaskTrackComponent
        {
            ComponentIndex = 0,
            SourceBoundsX = 60, SourceBoundsY = 30,
            SourceBoundsWidth = 12, SourceBoundsHeight = 12,
            Samples = new List<ManualMaskTrackSample>()

            });

            SourceKeyframe = 42,
            SourceMaskFingerprint = fingerprint,

// Exercise actual disk persistence, not only FrameMaskProvider's memory
// snapshots. A manual PNG and Auto rectangles at the same ordinal must
// round-trip independently, including intentionally overlapping pixels.
string stateRoot = Path.Combine(Path.GetTempPath(), "faceshield-mask-state-" + Guid.NewGuid().ToString("N"));
Directory.CreateDirectory(stateRoot);
try
{
    string video = Path.Combine(stateRoot, "source.mp4");
    File.WriteAllText(video, "source identity fixture");
    var diskStore = new WorkspaceStateStore(Path.Combine(stateRoot, "state"));
    const int diskFrame = 5;
    using var original = new FrameMaskProvider();
    original.SetFaceRects(diskFrame, new[] { autoFace }, size, 0.9f, new[] { 0.9f });
    using (var manual = ManualMaskEditorLayer.CreateEditableMask(original, diskFrame, size))
    {
        PaintRect(manual, 32, 36, 6, 6, 255);
        PaintRect(manual, 112, 28, 24, 24, 255);
        original.SetIndependentManualMask(diskFrame, Clone(manual));
    }

    var diskSnapshot = new WorkspaceSnapshot(video, WorkspaceMode.Auto, diskFrame,
        0, 5, DateTimeOffset.UtcNow, 0, true, null, true, true, null,
        false, false, null);
    diskStore.SaveWorkspace(diskSnapshot, original);

    using var restored = new FrameMaskProvider();
    var reopenedStore = new WorkspaceStateStore(Path.Combine(stateRoot, "state"));
    Assert(reopenedStore.TryLoadWorkspace(video, WorkspaceMode.Auto, restored, out var restoredSnapshot)
        && restoredSnapshot?.AutoCompleted == true, "Persisted workspace was not restored");
    Assert(restored.TryGetFaceMaskData(diskFrame, out var restoredAuto)
        && restoredAuto.Faces.Count == 1, "Disk persistence lost Auto layer on manual frame");
    using (var final = restored.GetFinalMask(diskFrame)
           ?? throw new InvalidOperationException("Restored union missing"))
        Assert(AlphaAt(final, 40, 38) > 0 && AlphaAt(final, 34, 38) == 255
            && AlphaAt(final, 120, 36) == 255, "Restored output lost Auto or manual coverage");

    restored.SetFaceRects(diskFrame, new[] { new Rect(54, 20, 26, 26) }, size);
    using (var final = restored.GetFinalMask(diskFrame)
           ?? throw new InvalidOperationException("Restored manual overlap missing"))
        Assert(AlphaAt(final, 34, 38) == 255 && AlphaAt(final, 40, 38) == 0
            && AlphaAt(final, 66, 32) > 0 && AlphaAt(final, 120, 36) == 255,
            "Restoring a manual PNG subtracted overlapping Auto pixels");

    // Both missing and unreadable PNGs must reject an incomplete generation.
    // The live provider and persisted metadata must not be overwritten.
    string png = Directory.GetFiles(Path.Combine(stateRoot, "state"),
        $"mask_{diskFrame}.png", SearchOption.AllDirectories).Single();
    string metadata = Path.Combine(stateRoot, "state", "state.json");
    string before = File.ReadAllText(metadata);
    // Invalid declared Auto entries must not disappear during cloning or
    // application. Exercise both rejection and complete backup recovery.
    string backup = metadata + ".bak";
    if (File.Exists(backup)) File.Delete(backup);
    var mutations = new Action<JsonObject>[]
    {
        ws => ws["FaceMasks"]!.AsArray()[0] = null,
        ws => ws["FaceMasks"]![0]!["Faces"]!.AsArray()[0] = null,
        ws => ws["FaceMasks"]![0]!["Faces"]![0]!["Width"] = 0,
        ws => ws["FaceMasks"]![0]!["FrameIndex"] = -1,
        ws => ws["FaceMasks"]![0]!["Width"] = width + 1,
        ws => ws["FaceMasks"]![0]!["Faces"]![0]!["X"] = width + 1,
        ws => ws["FaceMasks"]!.AsArray().Add(ws["FaceMasks"]![0]!.DeepClone()),
        ws => ws["MaskIndices"]!.AsArray().Add(diskFrame),
        ws => ws["FaceMasks"] = null
    };
    foreach (var mutate in mutations)
    {
        var root = JsonNode.Parse(before)!;
        mutate(root["Workspaces"]![0]!.AsObject());
        string damaged = root.ToJsonString();
        File.WriteAllText(metadata, damaged);
        bool rejected = false;
        try { reopenedStore.TryLoadWorkspace(video, WorkspaceMode.Auto, restored, out _); }
        catch (InvalidDataException) { rejected = true; }
        Assert(rejected, "Invalid Auto/PNG metadata was accepted as complete");
        Assert(File.ReadAllText(metadata) == damaged, "Rejected metadata was rewritten");
        using var unchanged = restored.GetFinalMask(diskFrame)
            ?? throw new InvalidOperationException("Rejected metadata cleared live masks");
        Assert(AlphaAt(unchanged, 66, 32) > 0 && AlphaAt(unchanged, 34, 38) == 255,
            "Rejected metadata changed live coverage");
    }
    File.WriteAllText(backup, before);
    using (var recovered = new FrameMaskProvider())
    {
        Assert(reopenedStore.TryLoadWorkspace(video, WorkspaceMode.Auto, recovered, out var recovery)
            && recovery?.AutoCompleted == true, "Complete backup did not recover malformed primary");
        Assert(recovered.TryGetFaceMaskData(diskFrame, out var recoveredAuto)
            && recoveredAuto.Faces.Count == 1, "Backup recovery lost Auto coverage");
    }
    // Unreadable whole-state files must never be treated as a fresh install.
    foreach (string invalidState in new[] { "{", "null", "{}", "{\"Recents\":null,\"Workspaces\":[]}" })
    {
        File.WriteAllText(metadata, invalidState);
        File.WriteAllText(backup, "invalid backup");
        bool startupRejected = false;
        try { _ = new WorkspaceStateStore(Path.Combine(stateRoot, "state")); }
        catch (InvalidDataException) { startupRejected = true; }
        Assert(startupRejected, "Corrupt whole-state metadata was treated as a new workspace");
        bool writeRejected = false;
        try { reopenedStore.SaveWorkspace(diskSnapshot, original); }
        catch (InvalidDataException) { writeRejected = true; }
        Assert(writeRejected && File.ReadAllText(metadata) == invalidState &&
            File.ReadAllText(backup) == "invalid backup" && File.Exists(png),
            "Failed save overwrote corrupt state or deleted existing payload");
    }

    // Valid backup recovers the damaged primary and remains valid on the
    // next save. Missing both files with existing payload is also unsafe.
    File.WriteAllText(backup, before);
    File.WriteAllText(metadata, "{");
    var backupStore = new WorkspaceStateStore(Path.Combine(stateRoot, "state"));
    Assert(File.ReadAllText(metadata) == before, "Valid state backup was not restored");
    File.Delete(metadata);
    File.Delete(backup);
    bool orphanRejected = false;
    try { _ = new WorkspaceStateStore(Path.Combine(stateRoot, "state")); }
    catch (InvalidDataException) { orphanRejected = true; }
    Assert(orphanRejected && File.Exists(png), "Missing state with orphan payload was treated as fresh");
    File.WriteAllText(metadata, before);
    File.Delete(png);
    foreach (bool corrupt in new[] { false, true })
    {
        if (corrupt) File.WriteAllText(png, "invalid png");
        bool rejected = false;
        try
        {
            reopenedStore.TryLoadWorkspace(video, WorkspaceMode.Auto, restored, out _);
        }
        catch (InvalidDataException) { rejected = true; }
        Assert(rejected, "Missing/corrupt persisted mask was accepted as complete");
        Assert(File.ReadAllText(metadata) == before, "Rejected load rewrote saved metadata");
        using var unchanged = restored.GetFinalMask(diskFrame)
            ?? throw new InvalidOperationException("Rejected load cleared live masks");
        Assert(AlphaAt(unchanged, 34, 38) == 255 && AlphaAt(unchanged, 66, 32) > 0,
            "Rejected load altered live masks");
    }
}
finally
{
    Directory.Delete(stateRoot, recursive: true);
}

static void AssertTrackParity(
    IReadOnlyDictionary<int, IReadOnlyList<FaceTrackDetection>> frames,
    FaceTrackPostProcessOptions options)
{
    var expected = BuildReference(frames, options);
    var actual = new FaceTrackBuilder().Build(frames, options);
    Assert(expected.Count == actual.Count, "Track count changed");
    for (int i = 0; i < expected.Count; i++)
        Assert(expected[i].Id == actual[i].Id &&
            expected[i].Detections.SequenceEqual(actual[i].Detections),
            "Track identity or detection order changed");
}

        static IReadOnlyList<FaceTrack> BuildReference(
            IReadOnlyDictionary<int, IReadOnlyList<FaceTrackDetection>> detectionsByFrame,
            FaceTrackPostProcessOptions options,
            CancellationToken cancellationToken = default)
        {
            var tracks = new List<FaceTrack>();
            int nextTrackId = 1;

            foreach (var frame in detectionsByFrame.OrderBy(static x => x.Key))
            {
                cancellationToken.ThrowIfCancellationRequested();
                var frameDetections = frame.Value
                    .OrderByDescending(static x => x.Confidence)
                    .ToArray();
                if (frameDetections.Length == 0)
                    continue;

                var usedTracks = new HashSet<int>();
                foreach (var detection in frameDetections)
                {
                    cancellationToken.ThrowIfCancellationRequested();
                    FaceTrack? bestTrack = null;
                    double bestScore = double.NegativeInfinity;

                    foreach (var track in tracks)
                    {
                        cancellationToken.ThrowIfCancellationRequested();
                        if (usedTracks.Contains(track.Id))
                            continue;

                        var last = track.LastDetection;
                        if (last == null)
                            continue;

                        int gap = detection.FrameIndex - last.Value.FrameIndex;
                        if (gap <= 0 || gap > options.MaxTrackGap)
                            continue;

                        double score = ReferenceScore(last.Value.Bounds, detection.Bounds, gap, options);
                        if (score > bestScore)
                        {
                            bestScore = score;
                            bestTrack = track;
                        }
                    }

                    if (bestTrack == null || bestScore < options.MinTrackMatchScore)
                    {
                        bestTrack = new FaceTrack(nextTrackId++);
                        tracks.Add(bestTrack);
                    }
                    else
                    {
                        usedTracks.Add(bestTrack.Id);
                    }

                    bestTrack.Add(detection);
                }
            }

            return tracks;
        }


        static double ReferenceScore(Rect previous, Rect current, int gap, FaceTrackPostProcessOptions options)
        {
            double iou = FaceTrackBuilder.IoU(previous, current);
            double areaRatio = FaceTrackBuilder.GetAreaRatio(previous, current);
            if (areaRatio > options.MaxAreaChangeRatio || areaRatio < 1.0 / options.MaxAreaChangeRatio)
                return double.NegativeInfinity;

            double centerShift = FaceTrackBuilder.GetNormalizedCenterShift(previous, current);
            double allowedShift = options.MaxCenterShiftRatio * Math.Max(1, gap);
            if (centerShift > allowedShift)
                return double.NegativeInfinity;

            double continuityBonus = Math.Max(0.0, 1.0 - centerShift / Math.Max(allowedShift, 0.001));
            if (iou < options.MinTrackIou && continuityBonus < options.MinCenterContinuity)
                return double.NegativeInfinity;

            return (iou * 2.0) + continuityBonus - (gap - 1) * 0.08;
        }


static void PaintRect(WriteableBitmap mask, int x0, int y0, int w, int h, byte alpha)
{
    using var fb = mask.Lock();
    unsafe
    {
        byte* origin = (byte*)fb.Address;
        for (int y = Math.Max(0, y0); y < Math.Min(fb.Size.Height, y0 + h); y++)
        {
            byte* row = origin + y * fb.RowBytes;
            for (int x = Math.Max(0, x0); x < Math.Min(fb.Size.Width, x0 + w); x++)
            {
                int p = x * 4;
                row[p] = row[p + 1] = row[p + 2] = row[p + 3] = alpha;
            }
        }
    }
}

static WriteableBitmap Clone(WriteableBitmap source)
{
    var copy = new WriteableBitmap(source.PixelSize, source.Dpi, PixelFormat.Bgra8888, AlphaFormat.Premul);
    using var src = source.Lock();
    using var dst = copy.Lock();
    unsafe
    {
        byte* s = (byte*)src.Address;
        byte* d = (byte*)dst.Address;
        int bytes = Math.Min(src.RowBytes, dst.RowBytes);
        for (int y = 0; y < src.Size.Height; y++)
            Buffer.MemoryCopy(s + y * src.RowBytes, d + y * dst.RowBytes, dst.RowBytes, bytes);
    }
    return copy;
}

static byte AlphaAt(WriteableBitmap mask, int x, int y)
{
    using var fb = mask.Lock();
    unsafe { return *((byte*)fb.Address + y * fb.RowBytes + x * 4 + 3); }
}

static void Assert(bool condition, string message)
{
    if (!condition) throw new InvalidOperationException(message);
}

// Real production FFmpeg tracker + timeline preview/export against synthetic frames.
using Avalonia;
using Avalonia.Headless;
using Avalonia.Media.Imaging;
using Avalonia.Platform;
using Avalonia.Skia;
using FaceShield.Services.Video;
using System;
using System.Reflection;
using System.Threading;

if (args.Length != 1 || !System.IO.File.Exists(args[0]))
    throw new ArgumentException("Expected the generated video path.");
const int width = 320, height = 180, sourceFrame = 550;
AppBuilder.Configure<OverlayTestApp>().UseSkia()
    .UseHeadless(new AvaloniaHeadlessPlatformOptions { UseHeadlessDrawing = false })
    .SetupWithoutStarting();
FFmpegBootstrap.Initialize();
using var provider = new FrameMaskProvider();
var sourceMask = new WriteableBitmap(new PixelSize(width, height), new Vector(96, 96),
    PixelFormat.Bgra8888, AlphaFormat.Premul);
using (var framebuffer = sourceMask.Lock())
{
    unsafe
    {
        byte* pixels = (byte*)framebuffer.Address;
        for (int y = 0; y < height; y++)
        {
            byte* row = pixels + y * framebuffer.RowBytes;
            for (int x = 0; x < width; x++)
            {
                byte a = x >= 95 && x < 190 && y >= 70 && y < 125 ? (byte)255 : (byte)0;
                int p = x * 4;
                row[p] = row[p + 1] = row[p + 2] = row[p + 3] = a;
            }
        }
    }
}
provider.SetMask(sourceFrame, sourceMask);
foreach (int frame in new[] { sourceFrame - 1, sourceFrame, sourceFrame + 1, sourceFrame + 2 })
{
    // At frame 551 Auto overlaps the manual target AND extends beyond it.
    // A lossy subtraction would erase genuine manual pixels at (185,95).
    Rect[] faces = frame == sourceFrame + 1
        ? new[] { new Rect(9, 10, 14, 14), new Rect(178, 85, 32, 30) }
        : new[] { new Rect(9, 10, 14, 14) };
    provider.SetFaceRects(frame, faces, new PixelSize(width, height));
}

var assembly = typeof(FrameMaskProvider).Assembly;
const BindingFlags internalStatic = BindingFlags.NonPublic | BindingFlags.Static;
Type timeline = assembly.GetType("FaceShield.Services.Video.ManualMaskKeyframeTimeline", true)!;
Type tracker = assembly.GetType("FaceShield.Services.Video.ManualMaskTrackingService", true)!;
Type fingerprint = assembly.GetType("FaceShield.Services.Video.ManualMaskFingerprint", true)!;
Type selector = assembly.GetType("FaceShield.Services.Video.ManualTrackingSourceSelection", true)!;
MethodInfo trackForward = tracker.GetMethod("TrackForward", internalStatic)!;
MethodInfo selectManual = selector.GetMethod("TryCloneContinuation", internalStatic)!;
MethodInfo preview = timeline.GetMethod("TryCloneEffectiveKeyframeMask", internalStatic)!;
MethodInfo setSegment = timeline.GetMethod("SetTrackSegment", internalStatic)!;
MethodInfo export = timeline.GetMethod("CreateExportMaskLease", internalStatic)!;
Check((int)timeline.GetMethod("GetNextExplicitKeyframe", internalStatic)!
    .Invoke(null, new object[] { provider, sourceFrame, 0 })! == int.MaxValue,
    "An unrelated Auto frame ended the manual track");
timeline.GetMethod("Configure", internalStatic)!
    .Invoke(null, new object?[] { provider, true, args[0] });

object tracked = Track(sourceFrame, sourceFrame + 3, sourceMask);
Check((int)Read(tracked, "ProcessedFrames") == 2, "Initial manual track lost samples");
object initialSegment = Read(tracked, "Segment");
initialSegment.GetType().GetProperty("SourceMaskFingerprint")!
    .SetValue(initialSegment, Fingerprint(sourceMask));
setSegment.Invoke(null, new object[] { provider, initialSegment });

object?[] storedArgs = { provider, sourceFrame, null };
Check((bool)selectManual.Invoke(null, storedArgs)!, "Stored manual source missing");
using (var manual = (WriteableBitmap)storedArgs[2]!) AssertManualOnly(manual);
object?[] previewArgs = { provider, sourceFrame + 1, null, CancellationToken.None };
Check((bool)preview.Invoke(null, previewArgs)!, "Initial inherited preview missing");
using (var bitmap = (WriteableBitmap)previewArgs[2]!) AssertBoth(bitmap);

using (var lease = (IDisposable)export.Invoke(null, new object[] { provider })!)
{
    object view = lease.GetType().GetProperty("Provider", BindingFlags.NonPublic | BindingFlags.Instance)!
        .GetValue(lease)!;
    MethodInfo getMask = view.GetType().GetMethod("GetFinalMask")!;
    using (var before = (WriteableBitmap?)getMask.Invoke(view, new object[] { sourceFrame - 1 }))
        Check(before != null && AlphaAt(before, 16, 17) > 0, "Auto before manual start lost");
    foreach (int frame in new[] { sourceFrame + 1, sourceFrame + 2 })
    {
        using var bitmap = (WriteableBitmap?)getMask.Invoke(view, new object[] { frame });
        Check(bitmap != null, "Tracked export mask missing");
        AssertBoth(bitmap!);
    }
    object?[] borrowed = { sourceFrame + 1, null };
    Check((bool)view.GetType().GetMethod("TryGetBorrowedStoredMask")!.Invoke(view, borrowed)!,
        "Borrowed export mask missing");
    AssertBoth((WriteableBitmap)borrowed[1]!);
    try
    {
        getMask.Invoke(view, new object[] { sourceFrame + 3 });
        throw new InvalidOperationException("Untracked mask was silently held");
    }
    catch (TargetInvocationException ex) when (ex.InnerException is InvalidOperationException) { }
}

// The source frame 551 has Auto but no explicit manual bitmap. Continuing
// must select only the validated inherited manual component, then preserve
// Auto in preview/export after promoting that source to a correction keyframe.
object?[] continuationArgs = { provider, sourceFrame + 1, null };
Check((bool)selectManual.Invoke(null, continuationArgs)!, "Manual continuation unavailable");
WriteableBitmap? continuedMask = (WriteableBitmap)continuationArgs[2]!;
try
{
    AssertManualOnly(continuedMask);
    Check(AlphaAt(continuedMask, 185, 95) > 0 &&
          AlphaAt(continuedMask, 200, 95) == 0,
        "Overlapping manual pixels were subtracted or Auto-only pixels contaminated source");
    Check(Array.IndexOf(provider.GetStoredMaskFrameIndices(), sourceFrame + 1) < 0,
        "Continuation selection wrote a keyframe prematurely");
    object continued = Track(sourceFrame + 1, sourceFrame + 3, continuedMask);
    Check((int)Read(continued, "ProcessedFrames") == 1, "Continuation lost next-frame sample");
    object segment = Read(continued, "Segment");
    segment.GetType().GetProperty("SourceMaskFingerprint")!
        .SetValue(segment, Fingerprint(continuedMask));
    typeof(FrameMaskProvider).GetMethod("SetIndependentManualMask",
        BindingFlags.NonPublic | BindingFlags.Instance)!
        .Invoke(provider, new object[] { sourceFrame + 1, continuedMask });
    continuedMask = null; // ownership transferred
    Check(provider.TryCloneStoredMask(sourceFrame + 1, out var retained),
        "Promoted manual correction was not stored");
    using (retained)
        Check(AlphaAt(retained, 185, 95) > 0 && AlphaAt(retained, 200, 95) == 0,
            "Promotion stripped overlapping manual pixels or stored Auto-only pixels");
    setSegment.Invoke(null, new object[] { provider, segment });
    object?[] nextArgs = { provider, sourceFrame + 2, null, CancellationToken.None };
    Check((bool)preview.Invoke(null, nextArgs)!, "Continued preview missing");
    using (var bitmap = (WriteableBitmap)nextArgs[2]!) AssertBoth(bitmap);
    using var lease = (IDisposable)export.Invoke(null, new object[] { provider })!;
    object view = lease.GetType().GetProperty("Provider", BindingFlags.NonPublic | BindingFlags.Instance)!
        .GetValue(lease)!;
    using var exported = (WriteableBitmap?)view.GetType().GetMethod("GetFinalMask")!
        .Invoke(view, new object[] { sourceFrame + 2 });
    Check(exported != null, "Continued export missing");
    AssertBoth(exported!);
}
finally { continuedMask?.Dispose(); }

object?[] missingArgs = { provider, sourceFrame + 3, null };
Check(!(bool)selectManual.Invoke(null, missingArgs)!,
    "An untracked frame was silently accepted as a source");
Check(provider.TryGetFaceMaskData(sourceFrame + 1, out _), "Auto data was mutated");
Console.WriteLine("PASS: overlapping Auto/manual independent continuation, correction persistence, FFmpeg tracking, preview/export, no untracked hold");

object Track(int start, int endExclusive, WriteableBitmap mask)
{
    try
    {
        return trackForward.Invoke(null, new object?[]
            { args[0], start, endExclusive, mask, null, CancellationToken.None })!;
    }
    catch (TargetInvocationException ex) when (ex.InnerException != null)
    {
        System.Runtime.ExceptionServices.ExceptionDispatchInfo.Capture(ex.InnerException).Throw();
        throw;
    }
}
string Fingerprint(WriteableBitmap mask) =>
    (string)fingerprint.GetMethod("Compute", internalStatic)!
        .Invoke(null, new object?[] { mask, CancellationToken.None })!;
static object Read(object result, string name) => result.GetType().GetProperty(name)!.GetValue(result)!;
static byte AlphaAt(WriteableBitmap bitmap, int x, int y)
{
    using var buffer = bitmap.Lock();
    unsafe { return *((byte*)buffer.Address + y * buffer.RowBytes + x * 4 + 3); }
}
static void AssertManualOnly(WriteableBitmap bitmap)
{
    Check(AlphaAt(bitmap, 130, 95) > 0, "Manual target missing from source");
    Check(AlphaAt(bitmap, 16, 17) == 0, "Unrelated Auto face contaminated tracking source");
}
static void AssertBoth(WriteableBitmap bitmap)
{
    Check(AlphaAt(bitmap, 130, 95) > 0, "Manual target missing from composite");
    Check(AlphaAt(bitmap, 16, 17) > 0, "Auto face missing from composite");
}
static void Check(bool condition, string message)
{
    if (!condition) throw new InvalidOperationException(message);
}
public sealed class OverlayTestApp : Application { }
